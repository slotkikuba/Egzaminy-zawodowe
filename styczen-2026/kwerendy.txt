1. SELECT marka, model, cena FROM pojazdy WHERE marka = "BM" ORDER BY cena DESC limit 15;
2. SELECT AVG(cena) AS "Średnia cena", MAX(cena) AS "Maksymalna cena" FROM pojazdy WHERE model LIKE "meta";
3. SELECT p.marka, p.model, p.cena, k.nazwa, k.doplata FROM pojazdy p JOIN kolory k ON k.id = p.kolor WHERE model LIKE "alfa";
4. SELECT marka, model, cena FROM pojazdy ORDER BY RAND() LIMIT 2;
